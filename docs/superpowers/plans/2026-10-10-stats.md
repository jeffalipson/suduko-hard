# Stats Implementation Plan (Plan 7 of 7)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Record every finished or abandoned game (spec §8.1) in SwiftData and show the stats page (spec §8.2): per mode × tier played/won/win %/best/average/streaks, a solve-time chart, assistance totals, and techniques beaten plus abandon rates.

**Architecture:** The numbers are computed by a pure `StatsCalculator` in `SudokuGame` over plain `GameRecordData` values, so it is unit-tested with `swift test`. The app stores records as a SwiftData `@Model GameRecord` (a thin mirror of `GameRecordData`), inserts one when a game is won or abandoned, and renders `StatsView` with Swift Charts.

**Tech Stack:** Swift 6.2, SwiftData, Swift Charts, SwiftUI, Swift Testing.

**Spec:** `docs/superpowers/specs/2026-10-10-hard-sudoku-design.md` (§8)

## Global Constraints

- Mistakes are recorded only if mistake highlighting was on during the game; otherwise `nil` and excluded from mistake stats.
- Clean solve = won with zero hints, checks, reveals and assists.
- Abandoned = starting a new game in the same mode × tier while one is unfinished.
- Streak = consecutive wins in a mode × tier with no abandon in between.
- Start date is derived as end date minus play time (the session tracks play time, not wall-clock start). Ruled here.

## Review Focus

1. **Mistakes when highlighting was off** → `nil`, excluded from averages. Task 1 test.
2. **Streaks broken by an abandon** → current streak resets. Task 1 test.
3. **Empty stats** (no games) → zeros/dashes, no division by zero. Task 1 test.
4. **Abandon recording** → only when a resumable game exists and isn't complete. Task 2 code.

```swift
import Foundation
import SudokuCore

/// One finished or abandoned game, as stats see it.
public struct GameRecordData: Codable, Equatable, Sendable {
    public var mode: Mode
    public var tier: Tier
    public var puzzleID: String
    public var endDate: Date
    public var elapsed: Double
    public var won: Bool
    /// Nil when mistake highlighting was never on during the game.
    public var mistakes: Int?
    public var hints: [String: Int]
    public var checks: Int
    public var reveals: Int
    public var assists: Int
    public var clean: Bool
    public var techniques: [TechniqueID]

    public init(mode: Mode, tier: Tier, puzzleID: String, endDate: Date, elapsed: Double, won: Bool, mistakes: Int?,
                hints: [String: Int], checks: Int, reveals: Int, assists: Int, clean: Bool, techniques: [TechniqueID]) {
        self.mode = mode; self.tier = tier; self.puzzleID = puzzleID; self.endDate = endDate; self.elapsed = elapsed
        self.won = won; self.mistakes = mistakes; self.hints = hints; self.checks = checks; self.reveals = reveals
        self.assists = assists; self.clean = clean; self.techniques = techniques
    }

    public init(session: GameSession, won: Bool, endDate: Date) {
        let log = session.log
        self.init(mode: session.puzzle.mode, tier: session.puzzle.grade.tier, puzzleID: session.puzzle.id, endDate: endDate,
                  elapsed: session.elapsed, won: won, mistakes: log.highlightingUsed ? log.mistakes : nil,
                  hints: log.hints, checks: log.checks, reveals: log.reveals, assists: log.autoCandidates + log.cageHelper,
                  clean: won && log.isClean, techniques: session.puzzle.techniquesRequired)
    }

    /// Wall-clock start, derived from play time.
    public var startDate: Date { endDate.addingTimeInterval(-elapsed) }
}

public struct TierStats: Equatable, Sendable {
    public var played = 0
    public var won = 0
    public var winRate = 0.0
    public var bestTime: Double?
    public var averageTime: Double?
    public var currentStreak = 0
    public var bestStreak = 0
}

public struct AssistanceSummary: Equatable, Sendable {
    public var games = 0
    public var mistakes = 0
    public var mistakeGames = 0
    public var hints: [String: Int] = [:]
    public var checks = 0
    public var reveals = 0
    public var assists = 0
    public var cleanSolves = 0
}

public enum StatsCalculator {
    static func filtered(_ records: [GameRecordData], mode: Mode, tier: Tier? = nil) -> [GameRecordData] {
        records.filter { $0.mode == mode && (tier == nil || $0.tier == tier) }.sorted { $0.endDate < $1.endDate }
    }

    public static func tier(_ records: [GameRecordData], mode: Mode, tier: Tier) -> TierStats {
        let list = filtered(records, mode: mode, tier: tier)
        var s = TierStats()
        s.played = list.count
        let wins = list.filter(\.won)
        s.won = wins.count
        s.winRate = list.isEmpty ? 0 : Double(wins.count) / Double(list.count)
        s.bestTime = wins.map(\.elapsed).min()
        s.averageTime = wins.isEmpty ? nil : wins.map(\.elapsed).reduce(0, +) / Double(wins.count)
        var run = 0
        for r in list {
            run = r.won ? run + 1 : 0
            s.bestStreak = max(s.bestStreak, run)
        }
        s.currentStreak = run
        return s
    }

    public static func assistance(_ records: [GameRecordData], mode: Mode) -> AssistanceSummary {
        var a = AssistanceSummary()
        for r in filtered(records, mode: mode) {
            a.games += 1
            if let m = r.mistakes {
                a.mistakes += m
                a.mistakeGames += 1
            }
            for (k, v) in r.hints { a.hints[k, default: 0] += v }
            a.checks += r.checks
            a.reveals += r.reveals
            a.assists += r.assists
            if r.clean { a.cleanSolves += 1 }
        }
        return a
    }

    /// For each technique, how many won puzzles needed it (most first).
    public static func techniques(_ records: [GameRecordData], mode: Mode) -> [(technique: TechniqueID, count: Int)] {
        var counts: [TechniqueID: Int] = [:]
        for r in filtered(records, mode: mode) where r.won { for t in r.techniques { counts[t, default: 0] += 1 } }
        return counts.map { ($0.key, $0.value) }.sorted { ($0.count, $1.technique.weight) > ($1.count, $0.technique.weight) }
    }

    /// Abandon rate per tier, highest first (tiers with no games left out).
    public static func abandonRates(_ records: [GameRecordData], mode: Mode) -> [(tier: Tier, rate: Double)] {
        Tier.allCases.compactMap { t -> (Tier, Double)? in
            let list = filtered(records, mode: mode, tier: t)
            guard !list.isEmpty else { return nil }
            return (t, Double(list.filter { !$0.won }.count) / Double(list.count))
        }
        .sorted { $0.1 > $1.1 }
    }

    /// Won games' solve times in date order, for the progress chart.
    public static func progress(_ records: [GameRecordData], mode: Mode, tier: Tier) -> [(date: Date, seconds: Double)] {
        filtered(records, mode: mode, tier: tier).filter(\.won).map { ($0.endDate, $0.elapsed) }
    }
}
```

```swift
import Foundation
import SwiftData
import SudokuCore

/// Persisted mirror of `GameRecordData` (stored as JSON-friendly primitives).
@Model
final class GameRecord {
    var mode: String
    var tier: Int
    var puzzleID: String
    var endDate: Date
    var elapsed: Double
    var won: Bool
    var mistakes: Int?
    var hintsJSON: Data
    var checks: Int
    var reveals: Int
    var assists: Int
    var clean: Bool
    var techniques: [String]

    init(_ d: GameRecordData) {
        mode = d.mode.rawValue
        tier = d.tier.rawValue
        puzzleID = d.puzzleID
        endDate = d.endDate
        elapsed = d.elapsed
        won = d.won
        mistakes = d.mistakes
        hintsJSON = (try? JSONEncoder().encode(d.hints)) ?? Data()
        checks = d.checks
        reveals = d.reveals
        assists = d.assists
        clean = d.clean
        techniques = d.techniques.map(\.rawValue)
    }

    var data: GameRecordData? {
        guard let m = Mode(rawValue: mode), let t = Tier(rawValue: tier) else { return nil }
        return GameRecordData(mode: m, tier: t, puzzleID: puzzleID, endDate: endDate, elapsed: elapsed, won: won,
                              mistakes: mistakes, hints: (try? JSONDecoder().decode([String: Int].self, from: hintsJSON)) ?? [:],
                              checks: checks, reveals: reveals, assists: assists, clean: clean,
                              techniques: techniques.compactMap(TechniqueID.init(rawValue:)))
    }
}
```

```swift
import Charts
import SwiftData
import SwiftUI
import SudokuCore
import SudokuGame

struct StatsView: View {
    @Query(sort: \GameRecord.endDate) private var stored: [GameRecord]
    @Environment(\.dismiss) private var dismiss
    @State private var mode: Mode = .classic
    @State private var chartTier: Tier = .hard

    private var records: [GameRecordData] { stored.compactMap(\.data) }

    var body: some View {
        NavigationStack {
            List {
                Picker("Mode", selection: $mode) {
                    Text("Classic").tag(Mode.classic)
                    Text("Killer").tag(Mode.killer)
                }
                .pickerStyle(.segmented)
                .listRowBackground(Color.clear)

                Section("By difficulty") {
                    ForEach(Tier.allCases, id: \.self) { tier in tierRow(tier) }
                }

                Section("Progress") {
                    Picker("Tier", selection: $chartTier) {
                        ForEach(Tier.allCases, id: \.self) { Text($0.name.capitalized).tag($0) }
                    }
                    let points = StatsCalculator.progress(records, mode: mode, tier: chartTier)
                    if points.isEmpty {
                        Text("Win a \(chartTier.name) game to see your times.").foregroundStyle(.secondary)
                    } else {
                        Chart(points, id: \.date) { p in
                            LineMark(x: .value("Date", p.date), y: .value("Minutes", p.seconds / 60))
                            PointMark(x: .value("Date", p.date), y: .value("Minutes", p.seconds / 60))
                        }
                        .chartYAxisLabel("Minutes")
                        .frame(height: 180)
                    }
                }

                Section("Assistance") {
                    let a = StatsCalculator.assistance(records, mode: mode)
                    stat("Clean solves", a.games == 0 ? "—" : "\(a.cleanSolves) of \(a.games)")
                    stat("Mistakes (highlighting on)", a.mistakeGames == 0 ? "—" : "\(a.mistakes) in \(a.mistakeGames) games")
                    stat("Hints", ["nudge", "spotlight", "lesson", "showMe"].map { "\(a.hints[$0, default: 0])" }.joined(separator: " / "))
                    stat("Checks · Reveals · Assists", "\(a.checks) · \(a.reveals) · \(a.assists)")
                }

                Section("Techniques you've beaten") {
                    let t = StatsCalculator.techniques(records, mode: mode)
                    if t.isEmpty { Text("None yet.").foregroundStyle(.secondary) }
                    ForEach(t, id: \.technique) { stat($0.technique.displayName, "\($0.count)") }
                }

                Section("Most abandoned") {
                    let rates = StatsCalculator.abandonRates(records, mode: mode).filter { $0.rate > 0 }
                    if rates.isEmpty { Text("Nothing abandoned.").foregroundStyle(.secondary) }
                    ForEach(rates, id: \.tier) { stat($0.tier.name.capitalized, "\(Int(($0.rate * 100).rounded()))%") }
                }
            }
            .navigationTitle("Stats")
            .toolbar { Button("Done") { dismiss() } }
        }
    }

    private func tierRow(_ tier: Tier) -> some View {
        let s = StatsCalculator.tier(records, mode: mode, tier: tier)
        return VStack(alignment: .leading, spacing: 6) {
            Text(tier.name.capitalized).font(.headline)
            HStack {
                mini("Played", "\(s.played)")
                mini("Won", "\(s.won)")
                mini("Win %", s.played == 0 ? "—" : "\(Int((s.winRate * 100).rounded()))")
                mini("Best", s.bestTime.map(GameView.clock) ?? "—")
                mini("Avg", s.averageTime.map(GameView.clock) ?? "—")
                mini("Streak", "\(s.currentStreak)/\(s.bestStreak)")
            }
        }
        .padding(.vertical, 2)
    }

    private func mini(_ label: String, _ value: String) -> some View {
        VStack(spacing: 2) {
            Text(value).font(.subheadline.monospacedDigit()).lineLimit(1).minimumScaleFactor(0.7)
            Text(label).font(.caption2).foregroundStyle(.secondary)
        }
        .frame(maxWidth: .infinity)
    }

    private func stat(_ label: String, _ value: String) -> some View {
        HStack { Text(label); Spacer(); Text(value).foregroundStyle(.secondary).monospacedDigit() }
    }
}
```
